    public function contactStore(Request $request)
    {
        $request->validate([
            'name' => 'required|string|max:255',
            'phone' => 'required|numeric',
            'email' => 'required|email',
        ]);
        // dd($request->all());
        $contact = new Contact();
        $contact->name = $request->name;
        $contact->phone = $request->phone;
        $contact->email = $request->email;
        $contact->details = $request->details;

        $contact->save();

        $emailData = [
            'name' => $contact->name,
            'phone' => $contact->phone,
            'email' => $contact->email,
            'message' => $contact->details,
        ];
        // dd($emailData);
        try {
            Mail::send('mail.thankyou', $emailData, function ($message) use ($emailData) {
                $message->to($emailData['email'])
                    ->subject('Thank You!');
            });

            Mail::send('mail.inquirydata', ['data' => $emailData], function ($message) {
                // $message->to('Jeet@intelliworkz.tech')
                $message->to('allwincoldchainsolutions@gmail.com')
                    ->subject('New Inquiry For Contact');
            });
        } catch (Exception $e) {
            // Log the error for debugging purposes
            Log::error('Mail send failed: ' . $e->getMessage());
        }
        return redirect()->route('thankyou');
    }
